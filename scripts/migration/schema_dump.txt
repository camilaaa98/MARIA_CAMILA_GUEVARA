CREATE TABLE usuarios (
    id_usuario INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    apellido TEXT NOT NULL,
    correo TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    rol TEXT NOT NULL CHECK (rol IN ('administrador','instructor')),
    estado INTEGER DEFAULT 1,
    creado_en TEXT DEFAULT CURRENT_TIMESTAMP
)

CREATE TABLE sqlite_sequence(name,seq)

CREATE TABLE logs (
    id_log INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario INTEGER,
    accion TEXT NOT NULL,
    fecha TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
)

CREATE TABLE "asistencias" (id_asistencia INTEGER PRIMARY KEY AUTOINCREMENT, id_aprendiz INTEGER NOT NULL, id_usuario INTEGER NOT NULL, fecha TEXT NOT NULL, hora_entrada TEXT NOT NULL, hora_salida TEXT, tipo TEXT NOT NULL CHECK (tipo IN ('entrada','salida','completa')), observaciones TEXT, FOREIGN KEY (id_aprendiz) REFERENCES "aprendices_temp"(id_aprendiz), FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario))

CREATE TABLE "fichas" (
	"id_ficha"	INTEGER,
	"numero_ficha"	INTEGER NOT NULL UNIQUE,
	"id_programa"	INTEGER NOT NULL,
	"jornada"	TEXT NOT NULL,
	"estado"	TEXT DEFAULT 'ACTIVO',
	PRIMARY KEY("id_ficha" AUTOINCREMENT),
	FOREIGN KEY("id_programa") REFERENCES "programas_formacion"("id_programa")
)

CREATE TABLE "programas_formacion" (
	"id_programa"	INTEGER,
	"nombre_programa"	TEXT NOT NULL,
	"nivel_formacion"	TEXT NOT NULL,
	PRIMARY KEY("id_programa" AUTOINCREMENT)
)

CREATE TABLE "aprendices" (
	"id_aprendiz"	INTEGER,
	"tipo_identificacion"	TEXT,
	"documento"	NUMERIC NOT NULL,
	"nombre"	TEXT NOT NULL,
	"apellido"	TEXT NOT NULL,
	"correo"	TEXT,
	"id_ficha"	INTEGER NOT NULL,
	"estado"	TEXT,
	"celular"	INTEGER,
	UNIQUE("documento","id_ficha"),
	PRIMARY KEY("id_aprendiz" AUTOINCREMENT),
	FOREIGN KEY("id_ficha") REFERENCES "fichas"("id_ficha")
)
